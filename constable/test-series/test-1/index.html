end; $$; create or replace function public.mcres_submit_test( p_attempt_id uuid, p_attempt_token text, p_answers jsonb ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_attempt public.mcres_english_attempts%rowtype; end; $$; revoke all on function public.mcres_start_test(text,text,text) from public; revoke all on function public.mcres_submit_test(uuid,text,jsonb) from public; grant execute on function public.mcres_start_test(text,text,text) to anon, authenticated;
